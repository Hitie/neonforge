cmake_minimum_required(VERSION 3.25)
project(NeonForge VERSION 0.1.1 LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "NEONFORGE currently targets Windows x64 only")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  add_compile_options(/W4 /permissive- /utf-8 /EHsc)
endif()
enable_testing()
add_library(nf_package STATIC src/json.cpp src/manifest.cpp)
target_include_directories(nf_package PUBLIC src include)
add_executable(nf_package_tests tests/package_tests.cpp)
target_link_libraries(nf_package_tests PRIVATE nf_package)
add_test(NAME package_behavior COMMAND nf_package_tests)
add_library(NeonForge.Core SHARED src/exports.cpp src/kernel.cpp)
target_link_libraries(NeonForge.Core PRIVATE nf_package)
add_executable(nf_kernel_tests tests/kernel_tests.cpp)
target_include_directories(nf_kernel_tests PRIVATE include)
target_link_libraries(nf_kernel_tests PRIVATE NeonForge.Core)
add_test(NAME kernel_behavior COMMAND nf_kernel_tests)
add_executable(nf_event_routes_tests tests/event_routes_test.cpp)
target_include_directories(nf_event_routes_tests PRIVATE src include)
add_test(NAME event_routes_behavior COMMAND nf_event_routes_tests)
add_executable(nf_benchmark benchmarks/runtime_benchmark.cpp)
target_link_libraries(nf_benchmark PRIVATE nf_package)
target_compile_definitions(nf_benchmark PRIVATE NF_BENCHMARK_CONFIG="$<CONFIG>")
add_executable(neonforge src/host.cpp)
target_link_libraries(neonforge PRIVATE nf_package)
add_library(nf_example SHARED examples/native/provider.c)
add_library(nf_consumer SHARED examples/native/consumer.cpp)
add_library(nf_failing SHARED tests/failing_plugin.c)
add_library(nf_noncancelling_runtime SHARED tests/noncancelling_runtime.cpp)
target_include_directories(nf_noncancelling_runtime PRIVATE include)
add_library(nf_c_header OBJECT tests/c_header_test.c)
foreach(target nf_example nf_consumer nf_failing nf_c_header)
  target_include_directories(${target} PRIVATE include)
endforeach()
find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
add_executable(nf_winmm_probe src/winmm_probe.cpp)
target_link_libraries(nf_winmm_probe PRIVATE nf_package)
if(Python3_FOUND)
  string(RANDOM LENGTH 12 ALPHABET 0123456789abcdef NF_FIXTURE_ID)
  set(NF_GENERATED_DIR "${CMAKE_BINARY_DIR}/generated-${NF_FIXTURE_ID}")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/tools/forge.py")
  execute_process(COMMAND ${Python3_EXECUTABLE} "${CMAKE_SOURCE_DIR}/tools/forge.py" new "${NF_GENERATED_DIR}" --id org.neonforge.generated --name "Generated ABI Check" RESULT_VARIABLE NF_GENERATED_RESULT OUTPUT_QUIET)
  if(NOT NF_GENERATED_RESULT EQUAL 0)
    message(FATAL_ERROR "Native project generator failed")
  endif()
  add_library(nf_generated SHARED "${NF_GENERATED_DIR}/src/plugin.c")
  target_include_directories(nf_generated PRIVATE include)
  add_executable(nf_plugin_abi_test tests/plugin_abi_test.cpp)
  target_include_directories(nf_plugin_abi_test PRIVATE include)
  add_test(NAME plugin_abi COMMAND nf_plugin_abi_test $<TARGET_FILE:nf_example> $<TARGET_FILE:nf_consumer> $<TARGET_FILE:nf_generated>)
  add_test(NAME host_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/host_integration.py $<TARGET_FILE_DIR:neonforge>)
  add_test(NAME tool_behavior COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/tool_tests.py)
  add_test(NAME winmm_probe COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/probe_test.py $<TARGET_FILE_DIR:neonforge>)
  add_test(NAME distribution_behavior COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/package_sdk_test.py $<TARGET_FILE_DIR:neonforge>)
  add_test(NAME benchmark_behavior COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/benchmark_test.py $<TARGET_FILE_DIR:neonforge>)
endif()
