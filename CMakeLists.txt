cmake_minimum_required(VERSION 3.25)
project(NeonForge VERSION 0.1.7 LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "NEONFORGE currently targets Windows x64 only")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  add_compile_options(/W4 /permissive- /utf-8 /EHsc)
endif()
enable_testing()
add_library(nf_package STATIC src/json.cpp src/manifest.cpp)
target_include_directories(nf_package PUBLIC src include)
add_library(nf_data_model STATIC src/data_values.cpp src/data_model.cpp src/data_project.cpp src/data_authoring.cpp src/data_session.cpp)
target_link_libraries(nf_data_model PUBLIC nf_package)
add_executable(nf_data_tests tests/data_tests.cpp)
target_link_libraries(nf_data_tests PRIVATE nf_data_model)
add_test(NAME data_behavior COMMAND nf_data_tests)
add_executable(nf_data src/data_cli.cpp)
target_link_libraries(nf_data PRIVATE nf_data_model)
add_executable(nf_package_tests tests/package_tests.cpp)
target_link_libraries(nf_package_tests PRIVATE nf_package)
add_test(NAME package_behavior COMMAND nf_package_tests)
add_library(NeonForge.Core SHARED src/exports.cpp src/kernel.cpp)
target_link_libraries(NeonForge.Core PRIVATE nf_package)
file(GLOB NF_LUA_SOURCES CONFIGURE_DEPENDS third_party/lua/*.c)
list(FILTER NF_LUA_SOURCES EXCLUDE REGEX "/lua(c)?\\.c$")
set_source_files_properties(${NF_LUA_SOURCES} PROPERTIES LANGUAGE CXX)
add_library(nf_lua STATIC ${NF_LUA_SOURCES})
target_include_directories(nf_lua PUBLIC third_party/lua)
if(MSVC)
  target_compile_options(nf_lua PRIVATE /W0)
endif()
add_library(nf_script STATIC src/script_host.cpp)
target_link_libraries(nf_script PUBLIC nf_lua nf_package)
add_executable(nf_script_tests tests/script_tests.cpp)
target_include_directories(nf_script_tests PRIVATE src include)
target_link_libraries(nf_script_tests PRIVATE nf_script NeonForge.Core)
add_test(NAME script_behavior COMMAND nf_script_tests)
set_tests_properties(script_behavior PROPERTIES TIMEOUT 30)
add_executable(nf_script_benchmark benchmarks/script_benchmark.cpp)
target_include_directories(nf_script_benchmark PRIVATE src include)
target_link_libraries(nf_script_benchmark PRIVATE nf_script NeonForge.Core)
add_executable(nf_kernel_tests tests/kernel_tests.cpp)
target_include_directories(nf_kernel_tests PRIVATE include)
target_link_libraries(nf_kernel_tests PRIVATE NeonForge.Core)
add_test(NAME kernel_behavior COMMAND nf_kernel_tests)
add_executable(nf_typed_tests tests/typed_tests.cpp)
target_include_directories(nf_typed_tests PRIVATE include)
target_link_libraries(nf_typed_tests PRIVATE NeonForge.Core)
add_test(NAME typed_behavior COMMAND nf_typed_tests)
add_executable(nf_typed_example examples/native/typed_calls.c)
target_include_directories(nf_typed_example PRIVATE include)
target_link_libraries(nf_typed_example PRIVATE NeonForge.Core)
add_test(NAME typed_example COMMAND nf_typed_example)
add_executable(nf_event_routes_tests tests/event_routes_test.cpp)
target_include_directories(nf_event_routes_tests PRIVATE src include)
add_test(NAME event_routes_behavior COMMAND nf_event_routes_tests)
add_executable(nf_benchmark benchmarks/runtime_benchmark.cpp)
target_link_libraries(nf_benchmark PRIVATE nf_package)
target_compile_definitions(nf_benchmark PRIVATE NF_BENCHMARK_CONFIG="$<CONFIG>")
add_executable(neonforge src/host.cpp)
target_link_libraries(neonforge PRIVATE nf_package nf_script)
add_library(nf_example SHARED examples/native/provider.c)
add_library(nf_consumer SHARED examples/native/consumer.cpp)
add_library(nf_failing SHARED tests/failing_plugin.c)
add_library(nf_noncancelling_runtime SHARED tests/noncancelling_runtime.cpp)
target_include_directories(nf_noncancelling_runtime PRIVATE include)
add_library(nf_c_header OBJECT tests/c_header_test.c)
foreach(target nf_example nf_consumer nf_failing nf_c_header)
  target_include_directories(${target} PRIVATE include)
endforeach()
find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
add_executable(nf_winmm_probe src/winmm_probe.cpp)
target_link_libraries(nf_winmm_probe PRIVATE nf_package)
if(Python3_FOUND)
  add_test(NAME data_cli_behavior COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/data_cli_test.py $<TARGET_FILE_DIR:neonforge>)

  add_test(NAME data_store_behavior COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/data_store_test.py $<TARGET_FILE_DIR:neonforge>)

  if(NOT NF_GENERATED_RESULT EQUAL 0)
    message(FATAL_ERROR "Native project generator failed")
  endif()
  add_library(nf_generated SHARED "${NF_GENERATED_DIR}/src/plugin.c")
  target_include_directories(nf_generated PRIVATE include)
  add_executable(nf_plugin_abi_test tests/plugin_abi_test.cpp)
  target_include_directories(nf_plugin_abi_test PRIVATE include)
  add_test(NAME plugin_abi COMMAND nf_plugin_abi_test $<TARGET_FILE:nf_example> $<TARGET_FILE:nf_consumer> $<TARGET_FILE:nf_generated>)
  add_test(NAME host_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/host_integration.py $<TARGET_FILE_DIR:neonforge>)
  add_test(NAME script_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/script_integration.py $<TARGET_FILE_DIR:neonforge>)

  add_test(NAME tool_behavior COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/tool_tests.py)

  add_test(NAME benchmark_behavior COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/benchmark_test.py $<TARGET_FILE_DIR:neonforge>)
endif()
